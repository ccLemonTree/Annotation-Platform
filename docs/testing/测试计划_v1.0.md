```python
# backend/tests/services/storage/test_local_storage.py

import pytest
from pathlib import Path
from io import BytesIO
import hashlib

class TestLocalStorage:
    
    async def test_save_file_basic(self, storage, sample_image):
        """测试基本文件保存"""
        result = await storage.save_file(sample_image, "test/image.jpg")
        assert result['path'] == "test/image.jpg"
        assert await storage.file_exists("test/image.jpg")
    
    async def test_copy_file_zero_copy(self, storage, sample_image, tmp_path):
        """⭐ 测试零拷贝（硬链接）"""
        await storage.save_file(sample_image, "source/image.jpg")
        success = await storage.copy_file("source/image.jpg", "dest/image.jpg")
        
        # 验证是硬链接或符号链接
        src_stat = (tmp_path / "source/image.jpg").stat()
        dst_stat = (tmp_path / "dest/image.jpg").stat()
        assert src_stat.st_ino == dst_stat.st_ino  # 硬链接
```

```python
@pytest.mark.integration
async def test_zero_copy_import(client: AsyncClient, storage_base_path):
    """验证零拷贝导入"""
    
    # 1. 记录磁盘使用量
    usage_before = get_disk_usage(storage_base_path)
    
    # 2. SmartEye下载100张图片
    task_id = await create_and_execute_smarteye_task(client)
    usage_after_download = get_disk_usage(storage_base_path)
    download_size = usage_after_download - usage_before
    
    # 3. 导入到3个不同项目
    for i in range(3):
        project_id = await create_project(client, f"项目{i}")
        await import_from_smarteye(client, project_id, task_id)
    
    usage_after_import = get_disk_usage(storage_base_path)
    import_overhead = usage_after_import - usage_after_download
    
    # 4. 验证：导入开销 < 1MB（仅元数据）
    assert import_overhead < 1 * 1024 * 1024
    
    print(f"✅ 零拷贝验证通过:")
    print(f"  下载大小: {download_size / 1024 / 1024:.2f} MB")
    print(f"  导入开销: {import_overhead / 1024:.2f} KB")
    print(f"  节省空间: {download_size * 2 / 1024 / 1024:.2f} MB")
```

```python
async def test_concurrent_smarteye_downloads(client: AsyncClient):
    """测试5个任务并发下载"""
    
    start_time = time.time()
    
    # 并发创建5个任务
    tasks = []
    for i in range(5):
        task = client.post("/api/smarteye/tasks", json={
            "task_name": f"性能测试{i}",
            "query_params": {...}
        })
        tasks.append(task)
    
    responses = await asyncio.gather(*tasks)
    task_ids = [r.json()['task_id'] for r in responses]
```

### 6.3 UAT缺陷等级

| 等级 | 定义 | 处理策略 |
|------|------|---------|
| P0 | 核心功能不可用、数据丢失 | 立即修复，阻塞上线 |
| P1 | 重要功能异常、性能不达标 | 优先修复，可延期上线 |

## 7. 安全测试

### 7.1 认证与授权

| 测试项 | 测试内容 | 预期结果 |
|--------|---------|----------|
| 未授权访问 | 不携带Token访问API | 返回401 |

| 越权访问 | 访问其他租户的项目 | 返回403 |
| SQL注入 | 参数中注入SQL | 参数化查询，注入无效 |
| XSS攻击 | 输入框注入脚本 | 自动转义，不执行 |

### 7.2 租户隔离测试

```python
@pytest.mark.security
async def test_tenant_isolation(client: AsyncClient):
    """租户A不能访问租户B的数据"""
    
    # 创建租户A的项目
    token_a = await get_tenant_token(client, tenant="A")
    project_a = await create_project(client, token_a)

    )
```
